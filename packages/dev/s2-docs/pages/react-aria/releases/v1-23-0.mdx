# v1.23.0

This release adds support for editable [Slider](../Slider#editable) values via a slotted NumberField.

```tsx render hideCode
"use client";
import {Input, NumberField, Slider, SliderFill, SliderThumb, SliderTrack} from 'react-aria-components';
import {Label} from 'vanilla-starter/Form';
import 'vanilla-starter/Slider.css';

<Slider
  defaultValue={0.5}
  maxValue={1}
  step={0.01}
  formatOptions={{style: 'percent'}}>
  <Label>Opacity</Label>
  <SliderTrack>
    <div className="track inset">
      <SliderFill />
    </div>
    <SliderThumb className="react-aria-SliderThumb indicator" />
  </SliderTrack>
  <NumberField slot="0">
    <Input className="react-aria-Input inset" />
  </NumberField>
</Slider>
```

It also adds handle-only pointer dragging via `pointerDragSource: 'handle'` in [drag and drop](../dnd#drag-handle) for [GridList](../GridList#drag-and-drop), [Table](../Table#drag-and-drop), and [Tree](../Tree#drag-and-drop).

<ExampleSwitcher type="component" examples={['GridList', 'Table', 'Tree']}>

```tsx render hideCode dir="react-aria/releases" files={['packages/dev/s2-docs/pages/react-aria/PokemonCOMPONENT.tsx', 'packages/dev/s2-docs/pages/react-aria/DroppableCOMPONENT.tsx']}
"use client";
import {useDragAndDrop} from 'react-aria-components/useDragAndDrop';
import {PokemonCOMPONENT, Pokemon} from '../PokemonCOMPONENT';
import {DroppableCOMPONENT} from '../DroppableCOMPONENT';

function DraggableCOMPONENT() {
  let {dragAndDropHooks} = useDragAndDrop({
    getItems(keys, items: Pokemon[]) {
      return items.map(item => ({
        'text/plain': item.name,
        'pokemon': JSON.stringify(item)
      }));
    },
    pointerDragSource: 'handle'
  });

  return <PokemonCOMPONENT dragAndDropHooks={dragAndDropHooks} />;
}

<div style={{display: 'flex', gap: 12, flexWrap: 'wrap', width: '100%', justifyContent: 'center'}}>
  <DraggableCOMPONENT />
  <DroppableCOMPONENT />
</div>
```

</ExampleSwitcher>

Other highlights include focus events for collection items, description slot support for TreeItem, and numerous bug fixes across components and hooks.

As always, many thanks to all of our contributors!
