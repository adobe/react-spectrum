# v1.9.0

This release introduces editable value support for [Slider](../Slider#editable) and [RangeSlider](../RangeSlider#editable) via the new SliderInput component.

```tsx render hideCode
"use client";
import {Slider, SliderInput} from '@react-spectrum/s2/Slider';

<Slider
  label="Opacity"
  defaultValue={0.5}
  maxValue={1}
  step={0.01}
  formatOptions={{style: 'percent'}}>
  <SliderInput />
</Slider>
```

It also adds handle-only pointer dragging via `pointerDragSource: 'handle'` in [drag and drop](../dnd#drag-handle) for [ListView](../ListView#drag-and-drop), [TableView](../TableView#drag-and-drop), and [TreeView](../TreeView#drag-and-drop)!

<ExampleSwitcher type="component" examples={['ListView', 'TableView', 'TreeView']}>

```tsx render hideCode type="s2" dir="s2/releases" files={['packages/dev/s2-docs/pages/s2/EditableCOMPONENT.tsx']}
"use client";
import {useDragAndDrop} from '@react-spectrum/s2/useDragAndDrop';
import {useTreeData} from '@react-spectrum/s2/useTreeData';
import {EditableCOMPONENT, FileItem, defaultItems} from '../EditableCOMPONENT';

function DraggableCOMPONENT() {
  let tree = useTreeData({initialItems: defaultItems});

  let {dragAndDropHooks} = useDragAndDrop<FileItem>({
    getItems(keys, items: FileItem[]) {
      return items.map(item => ({
        'text/plain': item.name
      }));
    },
    onReorder(e) {
      if (e.target.dropPosition === 'before') {
        tree.moveBefore(e.target.key, e.keys);
      } else if (e.target.dropPosition === 'after') {
        tree.moveAfter(e.target.key, e.keys);
      }
    },
    pointerDragSource: 'handle'
  });

  let processItem = item => ({
    ...item.value,
    children: item.children.map(processItem)
  });

  return <EditableCOMPONENT items={tree.items.map(processItem)} dragAndDropHooks={dragAndDropHooks} />;
}
```

</ExampleSwitcher>

## AI Components

This release introduces the [LightboxCarousel](../ai-components#attachments-and-actions) component for previewing image attachments in full view, updates AIButton styling, improves Chat scroll anchoring, and fixes card attachment widths and `prose()` macro packaging.
