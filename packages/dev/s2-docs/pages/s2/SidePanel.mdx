# SidePanel

<PageDescription>{docs.exports.SidePanel.description}</PageDescription>

A `SidePanel` wraps a [SideNav](SideNav), along with any other app chrome, in a container that collapses down to a narrow icon rail. It renders its own collapse/expand toggle at the bottom of the panel, and animates its width between the two states.

The expanded width comes from the `styles` prop you pass to `SidePanel`.

```tsx render type="s2" files={['packages/dev/s2-docs/pages/s2/SidePanelApp.tsx']}
"use client";
import {SidePanel, SideNav, SideNavItem, SideNavItemContent, SideNavItemLink, SideNavSection, SideNavHeader, Text} from '@react-spectrum/s2/SideNav';
import {SidePanelApp} from './SidePanelApp';
import {style} from '@react-spectrum/s2/style' with {type: 'macro'};
import CCLibrary from '@react-spectrum/s2/icons/CCLibrary';
import Files from '@react-spectrum/s2/icons/Files';
import Images from '@react-spectrum/s2/icons/Images';
import Project from '@react-spectrum/s2/icons/Project';

<SidePanelApp defaultSelectedRoute="/files">
  {({selectedRoute}) => (
    /*- begin highlight -*/
    <SidePanel
      defaultCollapsed
      styles={style({gridArea: 'sidebar', width: 208, marginX: 8})}>
    {/*- end highlight -*/}
      <SideNav
        aria-label="Files"
        selectedRoute={selectedRoute}
        defaultExpandedKeys={['projects']}
        styles={style({width: 'full'})}>
        <SideNavItem href="/files" textValue="Files">
          <SideNavItemContent>
            <SideNavItemLink>
              <Files />
              <Text>Files</Text>
            </SideNavItemLink>
          </SideNavItemContent>
        </SideNavItem>
        <SideNavItem id="your-libraries" href="/your-libraries" textValue="Your Libraries">
          <SideNavItemContent>
            <SideNavItemLink>
              <CCLibrary />
              <Text>Your Libraries</Text>
            </SideNavItemLink>
          </SideNavItemContent>
          <SideNavItem href="/photos" textValue="Photos">
            <SideNavItemContent>
              <SideNavItemLink>
                <Images />
                <Text>Photos</Text>
              </SideNavItemLink>
            </SideNavItemContent>
          </SideNavItem>
        </SideNavItem>
        <SideNavSection>
          <SideNavHeader>Work</SideNavHeader>
          <SideNavItem id="projects" href="/projects" textValue="Projects">
            <SideNavItemContent>
              <SideNavItemLink>
                <Project />
                <Text>Projects</Text>
              </SideNavItemLink>
            </SideNavItemContent>
            <SideNavItem href="/projects/website" textValue="Website redesign">

## Collapsing

The collapsed state is uncontrolled by default. Use `defaultCollapsed` for the initial state, or `isCollapsed` together with `onCollapsedChange` to control it — for example to persist the user's preference, or to collapse the panel from elsewhere in the app.

```tsx render type="s2" files={['packages/dev/s2-docs/pages/s2/SidePanelApp.tsx']}
"use client";
import {SidePanel, SideNav, SideNavItem, SideNavItemContent, SideNavItemLink, Collection, Text} from '@react-spectrum/s2/SideNav';
import {SidePanelApp} from './SidePanelApp';
import {style} from '@react-spectrum/s2/style' with {type: 'macro'};
import File from '@react-spectrum/s2/icons/File';
import Files from '@react-spectrum/s2/icons/Files';
import Folder from '@react-spectrum/s2/icons/Folder';
import Images from '@react-spectrum/s2/icons/Images';
import {useState} from 'react';

///- begin collapse -///
interface Item {
  id: number;
  title: string;
  type: 'directory' | 'file';
  href?: string;
  icon?: typeof Folder;
  children?: Item[];
}
///- end collapse -///
let items: Item[] = [
  /*- begin highlight -*/
  {id: 1, title: 'Documents', type: 'directory', href: '/documents', icon: Files, children: [
  /*- end highlight -*/
    {id: 2, title: 'Project', type: 'directory', href: '/documents/project', children: [
      {id: 3, title: 'Notes', type: 'file', href: '/notes'},
      {id: 4, title: 'Budget', type: 'file', href: '/budget'}
    ]}
  ]},
  /*- begin highlight -*/
  {id: 5, title: 'Photos', type: 'directory', href: '/photos', icon: Images, children: [
  /*- end highlight -*/
    {id: 6, title: 'Image 1', type: 'file', href: '/image-1'},
    {id: 7, title: 'Image 2', type: 'file', href: '/image-2'}
  ]}
];

function CollapsibleSidePanel() {
  /*- begin highlight -*/
  let [isCollapsed, setCollapsed] = useState(true);
  /*- end highlight -*/

  return (
    <SidePanelApp
      defaultSelectedRoute="/notes"
      /*- begin highlight -*/
      isCollapsed={isCollapsed}
      onCollapsedChange={setCollapsed}>
      {/*- end highlight -*/}
      {({selectedRoute}) => (
        /*- begin highlight -*/
        <SidePanel
          isCollapsed={isCollapsed}
          onCollapsedChange={setCollapsed}
          styles={style({gridArea: 'sidebar', width: 208, marginX: 8})}>
        {/*- end highlight -*/}
          <SideNav
            aria-label="Files"
            selectedRoute={selectedRoute}
            items={items}
            styles={style({width: 'full'})}>
            {function renderItem(item) {
              let Icon = item.icon ?? (item.type === 'directory' ? Folder : File);
              return (
                <SideNavItem textValue={item.title} href={item.href}>
                  <SideNavItemContent>
                    <SideNavItemLink>
                      <Icon />
                      <Text>{item.title}</Text>
                    </SideNavItemLink>
                  </SideNavItemContent>
                  {item.children && <Collection items={item.children}>
                    {renderItem}
                  </Collection>}
                </SideNavItem>
              );
            }}
          </SideNav>
        </SidePanel>
      )}
    </SidePanelApp>
  );
}

<CollapsibleSidePanel />
```

## Animations

`SidePanel` and [SideNav](SideNav) animate in two steps, and never at the same time. The panel's width animates with a CSS transition, so the rest of your page reflows along with it and no extra markup is required. Its contents animate separately, with a [view transition](https://developer.mozilla.org/en-US/docs/Web/API/View_Transition_API): each `SideNavItem` row is wrapped in a [`<ViewTransition>`](https://react.dev/reference/react/ViewTransition) and each `SideNavHeader` is named in CSS, so rows slide to their new positions instead of jumping, and rows that come and go fade in and out.

The two are kept apart because a view transition animates a snapshot of an element towards its final geometry, which only reads well while the panel is standing still — a row that is also being resized ends up squashed or stretched against a layout that hasn't caught up. Labels are free to wrap while the panel is narrower than they are, too, so they can't be on screen during a width animation without reflowing everything below them. So collapsing empties the panel out first, in the same render that starts the width transition, and expanding fills it back in only once the panel has reached its full width.

That leaves only one direction that animates its contents, plus item disclosure, which happens at a fixed width:

<StaticTable
  headers={['Transition type', 'Description']}
  codeColumns={[0]}
  rows={[
    ['side-panel-expand', 'The panel filling back in, once it has finished widening.'],
    ['side-nav-item-collapse / side-nav-item-expand', "A single item's children being disclosed or hidden."]
  ]} />

Expanding the panel puts section headers back into the layout, which pushes every row below them down, so a header holds its fade until the second half of the animation rather than appearing on top of the rows still sliding past it. When an individual item is disclosed, headers cross fade normally because they are only moving.

Labels deliberately do not get their own `<ViewTransition>`. Naming a descendant would lift it out of its row's snapshot and animate it against the page rather than with its row, so it would drift independently of the row it belongs to.

<StaticTable
  headers={['Context value', 'Description']}
  codeColumns={[0]}
  rows={[
    ["isCollapsed", "Whether the panel's contents are collapsed. This trails the panel itself while it is expanding, since its contents are only brought back once it has finished widening."],
    ['setCollapsed', 'Collapses or expands the panel. Useful for building your own toggle, or for expanding the panel when a collapsed control is activated.']
  ]} />

Read `isCollapsed` from `SidePanelContext` to render custom content you place inside the panel in its collapsed and expanded states. Because it trails the width, content that is only rendered when expanded is never on screen while the panel is changing size, so it can't reflow as the panel narrows.

Both steps respect `prefers-reduced-motion` — the width transition and the view transitions are disabled when it is set.

```tsx render type="s2" files={['packages/dev/s2-docs/pages/s2/SidePanelApp.tsx']}
"use client";
import {SidePanel, SideNav, SideNavItem, SideNavItemContent, SideNavItemLink, Collection, Text} from '@react-spectrum/s2/SideNav';
import {SidePanelApp, AccountFooter} from './SidePanelApp';
import {style} from '@react-spectrum/s2/style' with {type: 'macro'};
import File from '@react-spectrum/s2/icons/File';
import Files from '@react-spectrum/s2/icons/Files';
import Folder from '@react-spectrum/s2/icons/Folder';
import Images from '@react-spectrum/s2/icons/Images';

///- begin collapse -///
interface Item {
  id: number;
  title: string;
  type: 'directory' | 'file';
  href?: string;
  icon?: typeof Folder;
  children?: Item[];
}
let items: Item[] = [
  {id: 1, title: 'Documents', type: 'directory', href: '/documents', icon: Files, children: [
    {id: 2, title: 'Project', type: 'directory', href: '/documents/project', children: [
      {id: 3, title: 'Notes', type: 'file', href: '/notes'},
      {id: 4, title: 'Budget', type: 'file', href: '/budget'}
    ]}
  ]},
  {id: 5, title: 'Photos', type: 'directory', href: '/photos', icon: Images, children: [
    {id: 6, title: 'Image 1', type: 'file', href: '/image-1'},
    {id: 7, title: 'Image 2', type: 'file', href: '/image-2'}
  ]}
];
///- end collapse -///

<SidePanelApp defaultSelectedRoute="/notes">
  {({selectedRoute}) => (
    <SidePanel defaultCollapsed styles={style({gridArea: 'sidebar', width: 208, marginX: 8})}>
      <SideNav
        aria-label="Files"
        selectedRoute={selectedRoute}
        items={items}
        styles={style({width: 'full'})}>
        {function renderItem(item) {
          let Icon = item.icon ?? (item.type === 'directory' ? Folder : File);
          return (
            <SideNavItem textValue={item.title} href={item.href}>
              <SideNavItemContent>
                <SideNavItemLink>
                  <Icon />
                  <Text>{item.title}</Text>
                </SideNavItemLink>
              </SideNavItemContent>
              {item.children && <Collection items={item.children}>
                {renderItem}
              </Collection>}
            </SideNavItem>
          );
        }}
      </SideNav>
      {/*- begin highlight -*/}
      <AccountFooter />
      {/*- end highlight -*/}
    </SidePanel>
  )}
</SidePanelApp>
```

`AccountFooter` reads `isCollapsed` from `SidePanelContext` and simply hides its name with `display: none` while the panel is collapsed. It needs no transition of its own: because `isCollapsed` trails the width when expanding, the name is only ever added or removed while the panel is standing still.
