## Content

`SideNav` follows the [Collection Components API](collections), accepting both static and dynamic collections. This example shows a dynamic collection, passing a list of objects to the `items` prop, and a recursive function to render the children.

```tsx render type="s2" files={['packages/dev/s2-docs/pages/s2/RoutedSideNav.tsx']}
"use client";
import {SideNav, SideNavItem, SideNavItemContent, SideNavItemLink, Collection} from '@react-spectrum/s2/SideNav';
import {RoutedSideNav} from './RoutedSideNav';
import {style} from '@react-spectrum/s2/style' with {type: 'macro'};

///- begin collapse -///
interface Item {
  id: number;
  title: string;
  type: 'directory' | 'file';
  href?: string;
  children?: Item[];
}

let items: Item[] = [
  {id: 1, title: 'Documents', type: 'directory', children: [
    {id: 2, title: 'Project', type: 'directory', children: [
      {id: 3, title: 'Notes', type: 'file', href: '/notes'},
      {id: 4, title: 'Budget', type: 'file', href: '/budget'}
    ]}
  ]},
  {id: 5, title: 'Photos', type: 'directory', children: [
    {id: 6, title: 'Image 1', type: 'file', href: '/image-1'},
    {id: 7, title: 'Image 2', type: 'file', href: '/image-2'}
  ]}
];
///- end collapse -///

<RoutedSideNav defaultSelectedRoute="/notes">
  {({selectedRoute}) => (
    <SideNav
      aria-label="Files"
      styles={style({height: 260, width: 210})}
      selectedRoute={selectedRoute}
      defaultExpandedKeys={[1, 2, 5]}
      items={items}>
      {function renderItem(item) {
        return (
          <SideNavItem textValue={item.title} href={item.href}>
            <SideNavItemContent>{item.href ? <SideNavItemLink>{item.title}</SideNavItemLink> : item.title}</SideNavItemContent>
            {/*- begin highlight -*/}
            {/* recursively render children */}
            {item.children && <Collection items={item.children}>
              {renderItem}
            </Collection>}
            {/*- end highlight -*/}
          </SideNavItem>
        );
      }}
    </SideNav>
  )}
</RoutedSideNav>
```

### Slots

`SideNavItemContent` supports icons, `Text`, [SideNavItemLink](#sidenavitemlink), [ActionMenu](ActionMenu), and [ActionButtonGroup](ActionButtonGroup) as children.

```tsx render type="s2" files={['packages/dev/s2-docs/pages/s2/RoutedSideNav.tsx']}
"use client";
import {SideNav, SideNavItem, SideNavItemContent, SideNavItemLink, Collection, Text} from '@react-spectrum/s2/SideNav';
import {RoutedSideNav} from './RoutedSideNav';
import {ActionMenu, MenuItem} from '@react-spectrum/s2/ActionMenu';
import Folder from '@react-spectrum/s2/icons/Folder';
import File from '@react-spectrum/s2/icons/File';
import Edit from '@react-spectrum/s2/icons/Edit';
import Delete from '@react-spectrum/s2/icons/Delete';
import {style} from '@react-spectrum/s2/style' with {type: 'macro'};

///- begin collapse -///
interface Item {
  id: number;
  title: string;
  type: 'directory' | 'file';
  href?: string;
  children?: Item[];
}
let items: Item[] = [
  {id: 1, title: 'Documents', type: 'directory', children: [
    {id: 2, title: 'Project', type: 'directory', children: [
      {id: 3, title: 'Notes', type: 'file', href: '/notes'},
      {id: 4, title: 'Budget', type: 'file', href: '/budget'}
    ]}
  ]},
  {id: 5, title: 'Photos', type: 'directory', children: [
    {id: 6, title: 'Image 1', type: 'file', href: '/image-1'},
    {id: 7, title: 'Image 2', type: 'file', href: '/image-2'}
  ]}
];
///- end collapse -///

<RoutedSideNav defaultSelectedRoute="/notes">
  {({selectedRoute}) => (
    <SideNav
      aria-label="Files"
      styles={style({height: 300, width: 210})}
      selectedRoute={selectedRoute}
      defaultExpandedKeys={[1, 2, 5]}
      items={items}>
      {function renderItem(item) {
        return (
          <SideNavItem textValue={item.title} href={item.href}>
            <SideNavItemContent>
              {/*- begin highlight -*/}
              {
                item.href ? <SideNavItemLink>
                  {item.type === 'directory' ? <Folder /> : <File />}
                  <Text>{item.title}</Text>
                </SideNavItemLink> : <>
                  {item.type === 'directory' ? <Folder /> : <File />}
                  <Text>{item.title}</Text>
                </>
              }
              {/*- end highlight -*/}
              <ActionMenu>
                <MenuItem>
                  <Edit />
                  <Text>Edit</Text>
                </MenuItem>
                <MenuItem>
                  <Delete />
                  <Text>Delete</Text>
                </MenuItem>
              </ActionMenu>
            </SideNavItemContent>
            {item.children && <Collection items={item.children}>
              {renderItem}
            </Collection>}
          </SideNavItem>
        );
      }}
    </SideNav>
  )}
</RoutedSideNav>
```

## Routing

SideNavs do not support an uncontrolled selection state, you are responsible for managing it through the `selectedRoute` prop. You may wire this up
to a router or other state management solution. The example below derives `selectedRoute` from the router's current location,
and provides React Aria's `RouterProvider` with the router's `navigate` function so that clicking an item performs a client side navigation.

If a SideNavItem has an `href`, then you must pass a `SideNavItemLink` as a child of the `SideNavItemContent`.

See Routing setup in [Getting Started](./getting-started) for more info.

```tsx render type="s2" files={['packages/dev/s2-docs/pages/s2/router.tsx']}
"use client";
import {SideNav, SideNavItem, SideNavItemContent, SideNavItemLink, Text} from '@react-spectrum/s2/SideNav';
import {RouterProvider} from 'react-aria-components';
import {MemoryRouter, useLocation, useNavigate} from './router';
import {style} from '@react-spectrum/s2/style' with {type: 'macro'};
import CCLibrary from '@react-spectrum/s2/icons/CCLibrary';
import Files from '@react-spectrum/s2/icons/Files';
import Images from '@react-spectrum/s2/icons/Images';

function FilesSideNav() {
  /*- begin highlight -*/
  let navigate = useNavigate();
  let {pathname} = useLocation();

  return (
    <RouterProvider navigate={navigate}>
      <SideNav aria-label="Files" defaultExpandedKeys={['your-libraries']} selectedRoute={pathname} styles={style({height: 120, width: 210})}>
      {/*- end highlight -*/}
        <SideNavItem href="/files" textValue="Files">
          <SideNavItemContent>
            <SideNavItemLink>
              <Files />
              <Text>Files</Text>
            </SideNavItemLink>
          </SideNavItemContent>
        </SideNavItem>
        <SideNavItem id="your-libraries" href="/your-libraries" textValue="Your Libraries">
          <SideNavItemContent>
            <SideNavItemLink>
              <CCLibrary />
              <Text>Your Libraries</Text>
            </SideNavItemLink>
          </SideNavItemContent>
          <SideNavItem id="photos" href="/photos" textValue="Photos">
            <SideNavItemContent>
              <SideNavItemLink>
                <Images />
                <Text>Photos</Text>
              </SideNavItemLink>
            </SideNavItemContent>
          </SideNavItem>
        </SideNavItem>
      </SideNav>
    </RouterProvider>
  );
}

<MemoryRouter initialEntries={['/files']}>
  <FilesSideNav />
</MemoryRouter>
```

## SidePanel

A `SidePanel` wraps a `SideNav`, along with any other app chrome, in a container that collapses down to a narrow icon rail. It renders its own collapse/expand toggle at the bottom of the panel, and animates its width between the two states.

The expanded width comes from the `styles` prop you pass to `SidePanel`.

### Collapsing

The collapsed state is uncontrolled by default. Use `defaultCollapsed` for the initial state, or `isCollapsed` together with `onCollapsedChange` to control it — for example to persist the user's preference, or to collapse the panel from elsewhere in the app.

```tsx render type="s2" files={['packages/dev/s2-docs/pages/s2/SidePanelApp.tsx']}
"use client";
import {SidePanel, SideNav, SideNavItem, SideNavItemContent, SideNavItemLink, Collection, Text} from '@react-spectrum/s2/SideNav';
import {SidePanelApp} from './SidePanelApp';
import {style} from '@react-spectrum/s2/style' with {type: 'macro'};
import File from '@react-spectrum/s2/icons/File';
import Files from '@react-spectrum/s2/icons/Files';
import Folder from '@react-spectrum/s2/icons/Folder';
import Images from '@react-spectrum/s2/icons/Images';
import {useState} from 'react';

///- begin collapse -///
interface Item {
  id: number;
  title: string;
  type: 'directory' | 'file';
  href?: string;
  icon?: typeof Folder;
  children?: Item[];
}
///- end collapse -///
let items: Item[] = [
  /*- begin highlight -*/
  {id: 1, title: 'Documents', type: 'directory', href: '/documents', icon: Files, children: [
  /*- end highlight -*/
    {id: 2, title: 'Project', type: 'directory', href: '/documents/project', children: [
      {id: 3, title: 'Notes', type: 'file', href: '/notes'},
      {id: 4, title: 'Budget', type: 'file', href: '/budget'}
    ]}
  ]},
  /*- begin highlight -*/
  {id: 5, title: 'Photos', type: 'directory', href: '/photos', icon: Images, children: [
  /*- end highlight -*/
    {id: 6, title: 'Image 1', type: 'file', href: '/image-1'},
    {id: 7, title: 'Image 2', type: 'file', href: '/image-2'}
  ]}
];

function CollapsibleSidePanel() {
  /*- begin highlight -*/
  let [isCollapsed, setCollapsed] = useState(true);
  /*- end highlight -*/

  return (
    <SidePanelApp
      defaultSelectedRoute="/notes"
      /*- begin highlight -*/
      isCollapsed={isCollapsed}
      onCollapsedChange={setCollapsed}>
      {/*- end highlight -*/}
      {({selectedRoute}) => (
        /*- begin highlight -*/
        <SidePanel
          isCollapsed={isCollapsed}
          onCollapsedChange={setCollapsed}
          styles={style({gridArea: 'sidebar', width: 208, marginX: 8})}>
        {/*- end highlight -*/}
          <SideNav
            aria-label="Files"
            selectedRoute={selectedRoute}
            items={items}
            styles={style({width: 'full'})}>
            {function renderItem(item) {
              let Icon = item.icon ?? (item.type === 'directory' ? Folder : File);
              return (
                <SideNavItem textValue={item.title} href={item.href}>
                  <SideNavItemContent>
                    <SideNavItemLink>
                      <Icon />
                      <Text>{item.title}</Text>
                    </SideNavItemLink>
                  </SideNavItemContent>
                  {item.children && <Collection items={item.children}>
                    {renderItem}
                  </Collection>}
                </SideNavItem>
              );
            }}
          </SideNav>
        </SidePanel>
      )}
    </SidePanelApp>
  );
}

<CollapsibleSidePanel />
```

### Animations

`SidePanel` animates its own `width` between the expanded width you provide and the collapsed rail width, and `SideNav` animates its labels, chevrons, and nested rows in response. Custom components you place inside the panel should participate in the same transition, so the panel exposes its animation state on `SidePanelContext`.

<StaticTable
  headers={['Context value', 'Description']}
  codeColumns={[0]}
  rows={[
    ['isCollapsed', 'The target state of the panel. Flips as soon as the toggle is activated, so it leads the width transition when expanding and matches it when collapsing.'],

```tsx render type="s2" files={['packages/dev/s2-docs/pages/s2/SidePanelApp.tsx']}
"use client";
import {SidePanel, SideNav, SideNavItem, SideNavItemContent, SideNavItemLink, Collection, Text} from '@react-spectrum/s2/SideNav';
import {SidePanelApp, AccountFooter} from './SidePanelApp';
import {style} from '@react-spectrum/s2/style' with {type: 'macro'};

import Files from '@react-spectrum/s2/icons/Files';
import Folder from '@react-spectrum/s2/icons/Folder';
import Images from '@react-spectrum/s2/icons/Images';

///- begin collapse -///
interface Item {
  id: number;
  title: string;
  type: 'directory' | 'file';
  href?: string;
  icon?: typeof Folder;
  children?: Item[];
}
let items: Item[] = [
  {id: 1, title: 'Documents', type: 'directory', href: '/documents', icon: Files, children: [
    {id: 2, title: 'Project', type: 'directory', href: '/documents/project', children: [
      {id: 3, title: 'Notes', type: 'file', href: '/notes'},
      {id: 4, title: 'Budget', type: 'file', href: '/budget'}
    ]}
  ]},
  {id: 5, title: 'Photos', type: 'directory', href: '/photos', icon: Images, children: [
    {id: 6, title: 'Image 1', type: 'file', href: '/image-1'},
    {id: 7, title: 'Image 2', type: 'file', href: '/image-2'}
  ]}
];
///- end collapse -///

<SidePanelApp defaultSelectedRoute="/notes">
  {({selectedRoute}) => (
    <SidePanel defaultCollapsed styles={style({gridArea: 'sidebar', width: 208, marginX: 8})}>
      <SideNav
        aria-label="Files"
        selectedRoute={selectedRoute}
        items={items}
        styles={style({width: 'full'})}>
        {function renderItem(item) {
          let Icon = item.icon ?? (item.type === 'directory' ? Folder : File);
          return (
            <SideNavItem textValue={item.title} href={item.href}>
              <SideNavItemContent>
                <SideNavItemLink>
                  <Icon />
                  <Text>{item.title}</Text>
                </SideNavItemLink>
              </SideNavItemContent>
              {item.children && <Collection items={item.children}>
                {renderItem}
              </Collection>}
            </SideNavItem>
          );
        }}
      </SideNav>
      {/*- begin highlight -*/}
      <AccountFooter />
      {/*- end highlight -*/}
    </SidePanel>
  )}
</SidePanelApp>
```

`AccountFooter` hides its label with `display: none` while `isHidden` is true, so the label needs `transitionBehavior: 'allow-discrete'` to animate `display` alongside `opacity`, and `'@starting-style': 0` so the fade runs in the entering direction too. Without the starting style the label would pop straight to full opacity when the panel finishes widening.

#### Reduced motion

`SidePanel` and `SideNav` disable their transitions under `@media (prefers-reduced-motion: reduce)`. Custom content should do the same, either with the same media query in the `style` macro, or by skipping the transition entirely.

```tsx
const label = style({
  transition: {
    default: 'none',
    isReady: {
      default: '[opacity, display]',
      isHidden: 'none',
      /*- begin highlight -*/
      '@media (prefers-reduced-motion: reduce)': 'none'
      /*- end highlight -*/
    }
  },
  transitionDuration: 150
});
```
